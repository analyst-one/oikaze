Generating CSS Utilities from Oikaze

Oikaze can be used to generate CSS utility classes.

@use 'sass:map';
@use 'sass:string';

@use 'tokens' as tokens;

:root {
  @include tokens.css-definitions();
}

$hover: '&--hover:hover';
$focus: '&--focus:focus';

/* *** FONTS *** */

// A basic utility class for font family.
.font {
  @include tokens.variants('family') using ($token) {
    font-family: tokens.get($token);
  }
}

/* *** COLORS *** */

// A more advanced utility class for text color, with hover and focus pseudo-classes.
.color {
  @include tokens.variants('color', '&', $hover, $focus) using ($token) {
    --color-opacity: 1;
    color: tokens.alpha($token, var(--color-opacity, 1));
  }

  &-opacity {
    @include tokens.variants('$opacity', '&', $hover, $focus) using ($token) {
      --color-opacity: #{tokens.percentage($token)};
    }
  }
}

/* *** SIZES *** */

// We can also use the apply() function to generate utility classes from tokens that have multiple properties.
.text {
  @include tokens.variants(tokens.all('text', false)) using ($token) {
    @include tokens.apply($token);
  }
}
